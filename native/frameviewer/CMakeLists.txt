cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
project(MoodframeMedia VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Multimedia Test)
find_package(FFmpeg REQUIRED)
qt_standard_project_setup(REQUIRES 6.8)
add_library(moodframe_media_core STATIC Player.cpp Player.h
    vendor/media/Decoder.cpp vendor/media/Decoder.h
    vendor/media/MediaWorker.cpp vendor/media/MediaWorker.h
    vendor/media/VideoFrame.cpp vendor/media/VideoFrame.h
    vendor/media/AudioEngine.cpp vendor/media/AudioEngine.h
    vendor/media/PlaybackTrace.cpp vendor/media/PlaybackTrace.h)
target_include_directories(moodframe_media_core PUBLIC vendor)
target_compile_definitions(moodframe_media_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(moodframe_media_core PUBLIC Qt6::Core Qt6::Gui Qt6::Multimedia FFmpeg::FFmpeg d3d11 dxgi)
if(MSVC)
    target_compile_options(moodframe_media_core PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(MoodframeMedia MediaService.cpp)
target_compile_definitions(MoodframeMedia PRIVATE MOODFRAME_VERSION="${PROJECT_VERSION}")
target_link_libraries(MoodframeMedia PRIVATE moodframe_media_core)
install(TARGETS MoodframeMedia RUNTIME DESTINATION .)
file(GLOB ffmpeg_dlls "${FFMPEG_ROOT}/bin/*.dll")
install(FILES ${ffmpeg_dlls} DESTINATION .)
install(PROGRAMS "${FFMPEG_ROOT}/bin/ffmpeg.exe" DESTINATION .)
enable_testing()
qt_add_executable(MoodframePlayerTests Tests.cpp)
target_link_libraries(MoodframePlayerTests PRIVATE moodframe_media_core Qt6::Test)
add_test(NAME playback COMMAND MoodframePlayerTests -o "${CMAKE_CURRENT_BINARY_DIR}/test-results.txt,txt")
